using System.Collections;
using UnityEngine;
using TMPro;

/// <summary>
/// Controls floating visual feedback popups when collecting or sorting river waste.
/// Supports WCAG 2.2 accessibility through high-contrast visual cues.
/// </summary>
public class WastePopupVFX : MonoBehaviour
{
    // Generated by AI
    [Header("Animation Settings")]
    [SerializeField] private float floatSpeed = 1.5f;
    [SerializeField] private float fadeDuration = 0.8f;

    // Generated by AI
    [Header("UI References")]
    [SerializeField] private TextMeshProUGUI popupText;
    [SerializeField] private CanvasGroup canvasGroup;

    // Generated by AI
    /// <summary>
    /// Initializes the popup message, color contrast, and starts the fade-out coroutine.
    /// </summary>
    public void ShowPopup(string message, Color highContrastColor)
    {
        if (popupText != null)
        {
            popupText.text = message;
            popupText.color = highContrastColor;
        }

        StartCoroutine(AnimateAndFadeRoutine());
    }

    // Generated by AI
    private IEnumerator AnimateAndFadeRoutine()
    {
        float elapsedTime = 0f;

        while (elapsedTime < fadeDuration)
        {
            // Move the popup upward smoothly each frame
            transform.Translate(Vector3.up * floatSpeed * Time.deltaTime);

            // Gradually reduce alpha transparency for smooth fade-out
            if (canvasGroup != null)
            {
                canvasGroup.alpha = Mathf.Lerp(1f, 0f, elapsedTime / fadeDuration);
            }

            elapsedTime += Time.deltaTime;
            yield return null;
        }

        Destroy(gameObject);
    }
}
